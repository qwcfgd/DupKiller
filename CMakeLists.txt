cmake_minimum_required(VERSION 3.21)
project(QtDupKiller VERSION 1.0.0 LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "QtDupKiller currently supports Windows only")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(DUPKILLER_QT_MAJOR "6" CACHE STRING "Qt major version (5 or 6)")
set_property(CACHE DUPKILLER_QT_MAJOR PROPERTY STRINGS 5 6)
find_package(Qt${DUPKILLER_QT_MAJOR} REQUIRED COMPONENTS Core Widgets Concurrent Test)
set(QT Qt${DUPKILLER_QT_MAJOR})
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
if(MINGW)
  add_compile_options(-Wall -Wextra -Wpedantic -finput-charset=UTF-8 -fexec-charset=UTF-8)
elseif(MSVC)
  add_compile_options(/W4 /utf-8)
endif()
add_library(dup_core STATIC
  src/domain.cpp src/domain.h src/windowsfs.cpp src/windowsfs.h
  src/scanner.cpp src/scanner.h src/executor.cpp src/executor.h)
target_include_directories(dup_core PUBLIC src)
target_link_libraries(dup_core PUBLIC ${QT}::Core ${QT}::Concurrent PRIVATE ole32 shell32 uuid bcrypt)
add_library(dup_ui STATIC
  src/comparisonmodel.cpp src/comparisonmodel.h src/comparisonview.cpp src/comparisonview.h
  src/filetreemodel.cpp src/filetreemodel.h src/viewmodel.cpp src/viewmodel.h
  src/mainwindow.cpp src/mainwindow.h)
target_link_libraries(dup_ui PUBLIC dup_core ${QT}::Widgets)
add_executable(QtDupKiller WIN32 src/main.cpp resources/app.rc resources/icons.qrc)
target_link_libraries(QtDupKiller PRIVATE dup_ui)
add_executable(dup_task src/task.cpp)
target_link_libraries(dup_task PRIVATE dup_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(dup_benchmark tests/benchmark_ui.cpp)
  target_link_libraries(dup_benchmark PRIVATE dup_ui)
  add_executable(dup_fixture tests/fixture_ab.cpp)
  target_link_libraries(dup_fixture PRIVATE dup_ui)
  add_executable(dup_tests tests/test_dup.cpp resources/icons.qrc)
  target_link_libraries(dup_tests PRIVATE dup_ui ${QT}::Test)
  add_test(NAME dup_tests COMMAND dup_tests -o test-results.txt,txt -o test-results.xml,junitxml)
  # windeployqt patches Qt's runtime prefix. Keep test plugins available when
  # rebuilding a directory that already contains the deployed Qt DLLs.
  get_filename_component(DUPKILLER_QT_KIT "${${QT}_DIR}/../../.." ABSOLUTE)
  set_tests_properties(dup_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QPA_PLATFORM_PLUGIN_PATH=${DUPKILLER_QT_KIT}/plugins/platforms;QT_QPA_FONTDIR=C:/Windows/Fonts" TIMEOUT 180)
endif()
install(TARGETS QtDupKiller RUNTIME DESTINATION .)
